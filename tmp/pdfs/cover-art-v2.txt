 for cnt in contours:
  aa=cv2.approxPolyDP(cnt,.45,True).reshape(-1,2)

  pts=[((float(a[0])+dx)*sx,H-(float(a[1])+dy)*sy) for a in aa]
  p.moveTo(*pts[0])

 return p
